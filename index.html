// Ripples behind the hero const canvas = document.querySelector('.hero-ripples'); const ctx = canvas.getContext('2d'); const ripples = []; const LIFE = 4000; // each ripple lives 4 s let w = 0, h = 0, lastDrop = 0, raf = 0; function resize() { const dpr = window.devicePixelRatio || 1; // sharp lines on Retina screens w = canvas.clientWidth; h = canvas.clientHeight; canvas.width = w * dpr; canvas.height = h * dpr; ctx.setTransform(dpr, 0, 0, dpr, 0, 0); } function frame(t) { if (t - lastDrop > 900) { // a new drop every 0.9 s ripples.push({ x: Math.random() * w, y: Math.random() * h, born: t }); lastDrop = t; } while (ripples.length && t - ripples[0].born > LIFE) ripples.shift(); // drop expired ones ctx.clearRect(0, 0, w, h); ctx.strokeStyle = getComputedStyle(canvas).getPropertyValue('--ripple'); for (const r of ripples) { const age = (t - r.born) / LIFE; // 0 → 1 ctx.globalAlpha = 1 - age; // fade as it grows ctx.beginPath(); ctx.ellipse(r.x, r.y, age * 140, age * 140 * 0.38, 0, 0, Math.PI * 2); // flattened = water seen at an angle ctx.stroke(); } raf = requestAnimationFrame(frame); } if (!matchMedia('(prefers-reduced-motion: reduce)').matches) { resize(); addEventListener('resize', resize); // Only animate while the hero is on screen new IntersectionObserver(([entry]) => { cancelAnimationFrame(raf); if (entry.isIntersecting) raf = requestAnimationFrame(frame); }).observe(canvas); }
